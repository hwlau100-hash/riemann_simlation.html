<script>
  const canvas = document.getElementById('plotCanvas');
  const ctx = canvas.getContext('2d');
  const nSlider = document.getElementById('nSlider');
  const nDisplay = document.getElementById('nDisplay');
  const methodButtons = document.querySelectorAll('.btn-method');
  const valDx = document.getElementById('valDx');
  const valApprox = document.getElementById('valApprox');
  const valError = document.getElementById('valError');
  const statusTag = document.getElementById('statusTag');
  const formulaDetail = document.getElementById('formulaDetail');

  let currentN = 4;
  let currentMethod = 'left';

  const exactArea = 1 / 3;
  const f = (x) => x * x;

  methodButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      methodButtons.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      currentMethod = btn.dataset.method;
      render();
    });
  });

  nSlider.addEventListener('input', (e) => {
    currentN = parseInt(e.target.value, 10);
    nDisplay.textContent = currentN;
    render();
  });

  function setN(val) {
    currentN = val;
    nSlider.value = val;
    nDisplay.textContent = val;
    render();
  }

  function render() {
    const w = canvas.width;
    const h = canvas.height;
    const pad = 55;
    const plotW = w - 2 * pad;
    const plotH = h - 2 * pad;

    const toCanvasX = (x) => pad + x * plotW;
    const toCanvasY = (y) => (h - pad) - y * plotH;

    ctx.clearRect(0, 0, w, h);

    // Draw Grid & Axes
    ctx.strokeStyle = "#e2e8f0";
    ctx.lineWidth = 1;
    for (let i = 0; i <= 5; i++) {
      const v = i / 5;
      ctx.beginPath();
      ctx.moveTo(toCanvasX(v), toCanvasY(0));
      ctx.lineTo(toCanvasX(v), toCanvasY(1));
      ctx.stroke();

      ctx.beginPath();
      ctx.moveTo(toCanvasX(0), toCanvasY(v));
      ctx.lineTo(toCanvasX(1), toCanvasY(v));
      ctx.stroke();
    }

    ctx.strokeStyle = "#475569";
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(toCanvasX(0), toCanvasY(0));
    ctx.lineTo(toCanvasX(1.04), toCanvasY(0));
    ctx.moveTo(toCanvasX(0), toCanvasY(0));
    ctx.lineTo(toCanvasX(0), toCanvasY(1.04));
    ctx.stroke();

    // Axis Labels
    ctx.fillStyle = "#334155";
    ctx.font = "13px sans-serif";
    ctx.fillText("0", toCanvasX(0) - 14, toCanvasY(0) + 16);
    ctx.fillText("1", toCanvasX(1) - 4, toCanvasY(0) + 18);
    ctx.fillText("1", toCanvasX(0) - 18, toCanvasY(1) + 4);
    ctx.fillText("x", toCanvasX(1.04) + 4, toCanvasY(0) + 4);
    ctx.fillText("y", toCanvasX(0) - 4, toCanvasY(1.04) - 6);

    const dx = 1 / currentN;
    let approxSum = 0;

    // Draw Rectangles
    for (let i = 0; i < currentN; i++) {
      const xLeft = i * dx;
      const xRight = (i + 1) * dx;
      let evalX = xLeft;

      if (currentMethod === 'left') evalX = xLeft;
      else if (currentMethod === 'right') evalX = xRight;
      else if (currentMethod === 'midpoint') evalX = (xLeft + xRight) / 2;

      const height = f(evalX);
      approxSum += height * dx;

      const rectX = toCanvasX(xLeft);
      const rectY = toCanvasY(height);
      const rectW = plotW * dx;
      const rectH = (h - pad) - rectY;

      // Fill
      ctx.fillStyle = "rgba(37, 99, 235, 0.22)";
      ctx.fillRect(rectX, rectY, rectW, rectH);

      // Stroke
      ctx.strokeStyle = "#2563eb";
      ctx.lineWidth = 1.5;
      ctx.strokeRect(rectX, rectY, rectW, rectH);

      // Sample point marker
      ctx.fillStyle = "#d97706";
      ctx.beginPath();
      ctx.arc(toCanvasX(evalX), toCanvasY(height), 3.5, 0, Math.PI * 2);
      ctx.fill();
    }

    // Draw Curve y = x^2
    ctx.strokeStyle = "#dc2626";
    ctx.lineWidth = 2.5;
    ctx.beginPath();
    const steps = 150;
    for (let s = 0; s <= steps; s++) {
      const cx = s / steps;
      const cy = f(cx);
      const px = toCanvasX(cx);
      const py = toCanvasY(cy);
      if (s === 0) ctx.moveTo(px, py);
      else ctx.lineTo(px, py);
    }
    ctx.stroke();

    // Update Stats Card
    valDx.textContent = dx.toFixed(4);
    valApprox.textContent = approxSum.toFixed(6);
    const absErr = Math.abs(approxSum - exactArea);
    valError.textContent = absErr.toFixed(6);

    if (Math.abs(approxSum - exactArea) < 0.001) {
      statusTag.textContent = "Very Close (≈ Exact)";
      statusTag.className = "status-tag status-close";
    } else if (approxSum < exactArea) {
      statusTag.textContent = "Underestimate";
      statusTag.className = "status-tag status-under";
    } else {
      statusTag.textContent = "Overestimate";
      statusTag.className = "status-tag status-over";
    }

    // Update Algebraic Formula Details for S5 M2
    if (currentMethod === 'right') {
      formulaDetail.innerHTML = `<strong>Right Sum Formula (DSE M2):</strong><br>
        \\( R_n = \\sum_{i=1}^n \\left(\\frac{i}{n}\\right)^2 \\frac{1}{n} = \\frac{n(n+1)(2n+1)}{6n^3} = \\frac{1}{3} + \\frac{1}{2n} + \\frac{1}{6n^2} \\)<br>
        <em>Note the error \\( +\\frac{1}{2n} \\) confirms it is an overestimate.</em>`;
    } else if (currentMethod === 'left') {
      formulaDetail.innerHTML = `<strong>Left Sum Formula (DSE M2):</strong><br>
        \\( L_n = \\sum_{i=0}^{n-1} \\left(\\frac{i}{n}\\right)^2 \\frac{1}{n} = \\frac{(n-1)n(2n-1)}{6n^3} = \\frac{1}{3} - \\frac{1}{2n} + \\frac{1}{6n^2} \\)<br>
        <em>Note the error \\( -\\frac{1}{2n} \\) confirms it is an underestimate.</em>`;
    } else {
      formulaDetail.innerHTML = `<strong>Midpoint Sum Formula:</strong><br>
        \\( M_n = \\sum_{i=1}^n \\left(\\frac{2i-1}{2n}\\right)^2 \\frac{1}{n} = \\frac{1}{3} - \\frac{1}{12n^2} \\)<br>
        <em>Notice error decreases with \\( n^2 \\) rather than \\( n \\), converging significantly faster!</em>`;
    }
  }

  render();
</script>
